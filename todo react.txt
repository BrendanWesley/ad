import { useState } from 'react';
import { View, Text, TextInput, Button, FlatList } from 'react-native';

export default function App() {

  const [task, setTask] = useState('');
  const [list, setList] = useState([]);

  return (
    <View>
      <Text>To-Do App</Text>

      <TextInput placeholder="Enter task" onChangeText={setTask} />

      <Button
        title="Add"
        onPress={() => setList([...list, { id: Date.now().toString(), text: task }])}
      />

      <FlatList
        data={list}
        renderItem={({ item }) => (
          <Text onPress={() => setList(list.filter(i => i.id !== item.id))}>
            {item.text}
          </Text>
        )}
      />
    </View>
  );
}
